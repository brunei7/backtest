//@version=6
indicator('SMC Supply & Demand Probability Engine', overlay = true, max_bars_back = 500)

// ===================================================================
// --- Inputs ---
// ===================================================================
pivotLookback_str = input.string('1,2,3,5,11,15,20', title = 'Pivot Lookback Periods (comma-separated)', tooltip = 'Enter multiple lookback periods separated by a comma. Example: 5,15,30')
maxNumberOfBoxes = input.int(50, title = 'Max Number of Active Boxes per Type', minval = 1, maxval = 500)

// --- Display Settings ---
showBOS = input.bool(true, title = 'Show Break of Structure (BOS) Zones', group = 'Zone Display Settings')
showCHOCH = input.bool(true, title = 'Show Change of Character (CHOCH) Zones', group = 'Zone Display Settings')

// --- SMC Settings ---
requireInducement = input.bool(true, title = 'Require Inducement / Liquidity Grab', tooltip = 'If checked, a zone will only be drawn if the swing point creating it took liquidity from the previous swing point. This identifies high-probability zones.', group = 'SMC Settings')

// --- Alert Settings ---
showAlerts_NewPattern = input.bool(true, title = 'Alert on New Confirmed Zone', group = 'Alert Settings')
showAlerts_Mitigation = input.bool(true, title = 'Alert on Zone Mitigation', group = 'Alert Settings')
showAlerts_Break = input.bool(true, title = 'Alert on Zone Break', group = 'Alert Settings')


// ===================================================================
// --- Input Processing ---
// ===================================================================
var array<int> lookbackPeriods = array.new_int()
if barstate.isfirst
    array<string> lookback_str_array = str.split(pivotLookback_str, ',')
    for s in lookback_str_array
        array.push(lookbackPeriods, int(str.tonumber(str.trim(s))))

int numLookbacks = array.size(lookbackPeriods)


// ===================================================================
// --- Data Structures ---
// ===================================================================

// --- Bullish Boxes (Demand Zones) ---
var array<box> bullBoxes = array.new_box()
var array<label> bullLabels = array.new_label()
var array<float> bullBoxTops = array.new_float()
var array<float> bullBoxBottoms = array.new_float()
var array<bool> bullBoxMitigated = array.new_bool()

// --- Bearish Boxes (Supply Zones) ---
var array<box> bearBoxes = array.new_box()
var array<label> bearLabels = array.new_label()
var array<float> bearBoxTops = array.new_float()
var array<float> bearBoxBottoms = array.new_float()
var array<bool> bearBoxMitigated = array.new_bool()


// ===================================================================
// --- Structure Detection Logic (Anti-Repaint Upgrade V8.1) ---
// ===================================================================

// Swing Point Data for multiple Lookbacks
// Last High
var array<float> lastHighs = array.new_float(numLookbacks, na)
var array<int> lastHighBars = array.new_int(numLookbacks, na)
var array<float> lastHigh_open = array.new_float(numLookbacks, na)
var array<float> lastHigh_close = array.new_float(numLookbacks, na)
var array<float> lastHigh_highs = array.new_float(numLookbacks, na)
var array<float> lastHigh_lows = array.new_float(numLookbacks, na)

// Previous High
var array<float> prevHighs = array.new_float(numLookbacks, na)

// Last Low
var array<float> lastLows = array.new_float(numLookbacks, na)
var array<int> lastLowBars = array.new_int(numLookbacks, na)
var array<float> lastLows_open = array.new_float(numLookbacks, na)
var array<float> lastLows_close = array.new_float(numLookbacks, na)
var array<float> lastLows_high = array.new_float(numLookbacks, na)
var array<float> lastLows_low = array.new_float(numLookbacks, na)

// Previous Low
var array<float> prevLows = array.new_float(numLookbacks, na)

// Alert triggers for the current bar
var bool anyBullishBreakThisBar = false
var bool anyBearishBreakThisBar = false
anyBullishBreakThisBar := false
anyBearishBreakThisBar := false

// *** Anti-Repaint: semua update pivot digeser setelah candle confirmed ***
if barstate.isconfirmed
    for i = 0 to numLookbacks - 1 by 1
        int currentLookback = array.get(lookbackPeriods, i)

        // Identify Swing Points
        float pivotHighPrice = ta.pivothigh(currentLookback, currentLookback)
        float pivotLowPrice = ta.pivotlow(currentLookback, currentLookback)

        // Update High data
        if not na(pivotHighPrice)
            array.set(prevHighs, i, array.get(lastHighs, i))
            array.set(lastHighs, i, pivotHighPrice)
            array.set(lastHighBars, i, bar_index[currentLookback])
            array.set(lastHigh_open, i, open[currentLookback])
            array.set(lastHigh_close, i, close[currentLookback])
            array.set(lastHigh_highs, i, high[currentLookback])
            array.set(lastHigh_lows, i, low[currentLookback])

        // Update Low data
        if not na(pivotLowPrice)
            array.set(prevLows, i, array.get(lastLows, i))
            array.set(lastLows, i, pivotLowPrice)
            array.set(lastLowBars, i, bar_index[currentLookback])
            array.set(lastLows_open, i, open[currentLookback])
            array.set(lastLows_close, i, close[currentLookback])
            array.set(lastLows_high, i, high[currentLookback])
            array.set(lastLows_low, i, low[currentLookback])

        // Retrieve Swing Points
        float lastHigh = array.get(lastHighs, i)
        float prevHigh = array.get(prevHighs, i)
        float lastLow = array.get(lastLows, i)
        float prevLow = array.get(prevLows, i)

        // Break Conditions 
        bool isBullishBreak = close > lastHigh and high[1] < lastHigh
        bool isBearishBreak = close < lastLow and low[1] > lastLow
        bool inducementTaken_Bullish = lastLow < prevLow
        bool inducementTaken_Bearish = lastHigh > prevHigh

        // 📌 Zone Creation on Confirmed Break
        if isBullishBreak
            bool isCHOCH = lastHigh < prevHigh
            bool isBOS = na(prevHigh) or lastHigh > prevHigh

            if (isCHOCH and showCHOCH or isBOS and showBOS) and (not requireInducement or inducementTaken_Bullish)
                anyBullishBreakThisBar := true
                int lastLowBar = array.get(lastLowBars, i)

                float pivotLow_open = array.get(lastLows_open, i)
                float pivotLow_close = array.get(lastLows_close, i)
                float pivotLow_low = array.get(lastLows_low, i)
                float boxTop = math.max(pivotLow_open, pivotLow_close)
                float boxBottom = pivotLow_low

                b = box.new(lastLowBar, boxTop, lastLowBar + 5, boxBottom, border_color = color.new(color.green, 0), bgcolor = color.new(color.green, 80))
                l = label.new(lastLowBar, boxBottom, (isCHOCH ? 'CHOCH' : 'BOS') + ' Demand (' + str.tostring(currentLookback) + ')', xloc = xloc.bar_index, yloc = yloc.price, style = label.style_label_up, color = color.new(color.green, 20), textcolor = color.white, size = size.small)

                array.push(bullBoxes, b)
                array.push(bullLabels, l)
                array.push(bullBoxTops, boxTop)
                array.push(bullBoxBottoms, boxBottom)
                array.push(bullBoxMitigated, false)

        if isBearishBreak
            bool isCHOCH = lastLow > prevLow
            bool isBOS = na(prevLow) or lastLow < prevLow

            if (isCHOCH and showCHOCH or isBOS and showBOS) and (not requireInducement or inducementTaken_Bearish)
                anyBearishBreakThisBar := true
                int lastHighBar = array.get(lastHighBars, i)

                float pivotHigh_open = array.get(lastHigh_open, i)
                float pivotHigh_close = array.get(lastHigh_close, i)
                float pivotHigh_high = array.get(lastHigh_highs, i)
                float boxTop = pivotHigh_high
                float boxBottom = math.min(pivotHigh_open, pivotHigh_close)

                b = box.new(lastHighBar, boxTop, lastHighBar + 5, boxBottom, border_color = color.new(color.red, 0), bgcolor = color.new(color.red, 80))
                l = label.new(lastHighBar, boxTop, (isCHOCH ? 'CHOCH' : 'BOS') + ' Supply (' + str.tostring(currentLookback) + ')', xloc = xloc.bar_index, yloc = yloc.price, style = label.style_label_down, color = color.new(color.red, 20), textcolor = color.white, size = size.small)

                array.push(bearBoxes, b)
                array.push(bearLabels, l)
                array.push(bearBoxTops, boxTop)
                array.push(bearBoxBottoms, boxBottom)
                array.push(bearBoxMitigated, false)


// ===================================================================
// --- Box Management (Mitigation, Break, Delete) ---
// ===================================================================
if barstate.isconfirmed
    if array.size(bullBoxes) > 0
        for i = array.size(bullBoxes) - 1 to 0 by 1
            b_box = array.get(bullBoxes, i)
            b_label = array.get(bullLabels, i)
            b_top = array.get(bullBoxTops, i)
            b_bottom = array.get(bullBoxBottoms, i)
            b_mitigated = array.get(bullBoxMitigated, i)

            // Zone Broken
            if low < b_bottom
                box.delete(b_box)
                label.delete(b_label)
                array.remove(bullBoxes, i)
                array.remove(bullLabels, i)
                array.remove(bullBoxTops, i)
                array.remove(bullBoxBottoms, i)
                array.remove(bullBoxMitigated, i)
                continue

            // Zone Mitigated
            if not b_mitigated and low <= b_top
                box.set_border_color(b_box, color.gray)
                box.set_bgcolor(b_box, color.new(color.gray, 85))
                label.set_color(b_label, color.gray)
                array.set(bullBoxMitigated, i, true)

    if array.size(bearBoxes) > 0
        for i = array.size(bearBoxes) - 1 to 0 by 1
            b_box = array.get(bearBoxes, i)
            b_label = array.get(bearLabels, i)
            b_top = array.get(bearBoxTops, i)
            b_bottom = array.get(bearBoxBottoms, i)
            b_mitigated = array.get(bearBoxMitigated, i)

            // Zone Broken
            if high > b_top
                box.delete(b_box)
                label.delete(b_label)
                array.remove(bearBoxes, i)
                array.remove(bearLabels, i)
                array.remove(bearBoxTops, i)
                array.remove(bearBoxBottoms, i)
                array.remove(bearBoxMitigated, i)
                continue

            // Zone Mitigated
            if not b_mitigated and high >= b_bottom
                box.set_border_color(b_box, color.gray)
                box.set_bgcolor(b_box, color.new(color.gray, 85))
                label.set_color(b_label, color.gray)
                array.set(bearBoxMitigated, i, true)


// ===================================================================
// Max Box Limit
// ===================================================================
while array.size(bullBoxes) > maxNumberOfBoxes
    box.delete(array.get(bullBoxes, 0))
    label.delete(array.get(bullLabels, 0))
    array.shift(bullBoxes)
    array.shift(bullLabels)
    array.shift(bullBoxTops)
    array.shift(bullBoxBottoms)
    array.shift(bullBoxMitigated)

while array.size(bearBoxes) > maxNumberOfBoxes
    box.delete(array.get(bearBoxes, 0))
    label.delete(array.get(bearLabels, 0))
    array.shift(bearBoxes)
    array.shift(bearLabels)
    array.shift(bearBoxTops)
    array.shift(bearBoxBottoms)
    array.shift(bearBoxMitigated)


// ===================================================================
// 📌 ALERTS
// ===================================================================
alertcondition(showAlerts_NewPattern and anyBullishBreakThisBar, 'New Confirmed Demand Zone', 'New Demand Zone (BOS/CHOCH) confirmed on {{ticker}}.')
alertcondition(showAlerts_NewPattern and anyBearishBreakThisBar, 'New Confirmed Supply Zone', 'New Supply Zone (BOS/CHOCH) confirmed on {{ticker}}.')
alertcondition(showAlerts_Mitigation and anyBullishBreakThisBar, 'Zone Mitigated', 'Demand Zone mitigated on {{ticker}}.')
alertcondition(showAlerts_Mitigation and anyBearishBreakThisBar, 'Zone Mitigated', 'Supply Zone mitigated on {{ticker}}.')
alertcondition(showAlerts_Break and (anyBullishBreakThisBar or anyBearishBreakThisBar), 'Zone Broken', 'Zone Broken on {{ticker}}.')



// ===================================================================
// 🟢 MODE A+ — BUY ONLY + SCALE IN + DANGER AREA (FINAL)
// ===================================================================

// ===============================
// 🔧 USER TOGGLES
// ===============================
groupA = 'DASHBOARD'
showDashboard = input.bool(true, 'Show Dashboard', group = groupA)
showLevels = input.bool(true, 'Show Entry / SL / TP Lines', group = groupA)

// ===============================
// 🎯 TP SETTINGS (USER EDITABLE)
// ===============================
groupTP = 'TP SETTINGS'

// TP based on Fib multiplier
tp1_mult = input.float(0.5, 'TP1 Multiplier', step = 0.01, group = groupTP)
tp2_mult = input.float(0.618, 'TP2 Multiplier', step = 0.01, group = groupTP)
tp3_mult = input.float(0.786, 'TP3 Multiplier', step = 0.01, group = groupTP)

// Alternative: Risk Reward Mode
useRR_TP = input.bool(false, 'Use Risk : Reward Mode', group = groupTP)
tp1_rr = input.float(1.0, 'TP1 RR', step = 0.1, group = groupTP)
tp2_rr = input.float(2.0, 'TP2 RR', step = 0.1, group = groupTP)
tp3_rr = input.float(3.0, 'TP3 RR', step = 0.1, group = groupTP)

// ===============================
// 🧭 DASHBOARD SETTINGS
// ===============================
dashPosOpt = input.string('Top Right', 'Dashboard Position', options = ['Top Right', 'Top Left', 'Bottom Right', 'Bottom Left'], group = groupA)

dashSizeOpt = input.string('Medium', 'Dashboard Text Size', options = ['Tiny', 'Small', 'Medium', 'Standard', 'Large', 'Extra Large'], group = groupA)

// ===============================
// 🧠 DASHBOARD POSITION MAP
// ===============================
dashPos = dashPosOpt == 'Top Left' ? position.top_left : dashPosOpt == 'Bottom Right' ? position.bottom_right : dashPosOpt == 'Bottom Left' ? position.bottom_left : position.top_right

// ===============================
// 🔠 DASHBOARD TEXT SIZE MAP
// ===============================
dashTextSize = dashSizeOpt == 'Tiny' ? size.tiny : dashSizeOpt == 'Medium' ? size.small : dashSizeOpt == 'Standard' ? size.normal : dashSizeOpt == 'Large' ? size.large : dashSizeOpt == 'Extra Large' ? size.huge : size.small // default Small

// ===============================
// 🧠 TRADE STATE
// ===============================
var bool modeA_Valid = false
var bool modeA_Locked = false
var bool modeA_ScaleUsed = false

var int modeA_EntryBar = na
var float modeA_Entry = na
var float modeA_SL = na
var float modeA_TP1 = na
var float modeA_TP2 = na
var float modeA_TP3 = na
var int modeA_BOS_Count = 0

var float modeA_ScalePrice = na // harga scale in terakhir

// ===============================
// 🔒 LOCKED SMC RANGE (AT ENTRY)
// ===============================
var float modeA_smcHL = na
var float modeA_smcHH = na

// ===============================
// 🧠 STRUCTURE FLAGS (DARI SMC)
// ===============================
bool structureDemand = anyBullishBreakThisBar
bool structureSupply = anyBearishBreakThisBar

// ===============================
// 📦 DEMAND SNAPSHOT
// ===============================
float demandBottom = na
if array.size(bullBoxBottoms) > 0
    demandBottom := array.get(bullBoxBottoms, array.size(bullBoxBottoms) - 1)
    demandBottom

// ===============================
// 📐 SMC RANGE (HH–HL)
// ===============================
float smcHL = na
float smcHH = na

if array.size(bullBoxBottoms) > 0 and array.size(bearBoxTops) > 0
    smcHL := array.get(bullBoxBottoms, array.size(bullBoxBottoms) - 1)
    smcHH := array.get(bearBoxTops, array.size(bearBoxTops) - 1)
    smcHH

bool hitSL = modeA_Valid and low <= modeA_SL

// ===============================
// 🚨 SUPPLY NEAR ENTRY (READ-ONLY FOR ACCUM)
// ===============================
bool supplyNearEX_accum = false

if array.size(bearBoxBottoms) > 0 and not na(demandBottom)
    float tempRisk = math.abs(close - demandBottom)
    float tempDR = tempRisk * 0.25
    for i = 0 to array.size(bearBoxBottoms) - 1 by 1
        if math.abs(array.get(bearBoxBottoms, i) - close) <= tempDR
            supplyNearEX_accum := true
            supplyNearEX_accum

// ===============================
// 🧠 ACCUMULATION % ENGINE (READ-ONLY)
// ===============================

// --- Basic existence ---
bool hasSupply = array.size(bearBoxes) > 0
bool hasDemand = array.size(bullBoxes) > 0

int lastSupplyBar = hasSupply ? box.get_left(array.get(bearBoxes, array.size(bearBoxes) - 1)) : na
int lastDemandBar = hasDemand ? box.get_left(array.get(bullBoxes, array.size(bullBoxes) - 1)) : na

// --- Structural conditions ---
bool demandAfterSupply = hasSupply and hasDemand and lastDemandBar > lastSupplyBar

bool demandStillValid = false

if array.size(bullBoxBottoms) > 0
    demandStillValid := low > array.get(bullBoxBottoms, array.size(bullBoxBottoms) - 1)
    demandStillValid

// ===============================
// 🔢 ACCUM COMPONENTS (0–20 each)
// ===============================

// 1️⃣ Supply → Demand sequence
int acc_seq = demandAfterSupply ? 20 : 0

// 2️⃣ Demand validity
int acc_demand = demandStillValid ? 20 : 0

// 3️⃣ Time-based inventory building
int barsSinceDemand = hasDemand ? bar_index - lastDemandBar : 0
int acc_time = barsSinceDemand >= 10 ? 20 : barsSinceDemand >= 5 ? 15 : barsSinceDemand >= 2 ? 10 : 5

// 4️⃣ Supply pressure absence
int acc_supply = structureSupply ? 0 : supplyNearEX_accum ? 10 : 20

// 5️⃣ Pre-BOS structural support
int acc_structure = structureDemand ? 20 : modeA_BOS_Count >= 1 ? 10 : 5

// ===============================
// 🧮 FINAL ACCUMULATION %
// ===============================
int accumPercent = acc_seq + acc_demand + acc_time + acc_supply + acc_structure

// ===============================
// 🏦 INSTITUTIONAL ACCUM TEXT
// ===============================
string accumText = accumPercent < 20 ? 'NO INTEREST' : accumPercent < 35 ? 'PROBING' : accumPercent < 55 ? 'EARLY ACCUM' : accumPercent < 75 ? 'COMMIT' : 'EXPANSION'

color accumColor = accumPercent >= 80 ? color.lime : accumPercent >= 60 ? color.green : accumPercent >= 40 ? color.orange : color.gray

// ===============================
// 🔥 HARD RESET (SL / SUPPLY)
// ===============================
bool tp3Hit = modeA_Valid and high >= modeA_TP3

if hitSL or structureSupply or tp3Hit
    modeA_Valid := false
    modeA_Locked := false
    modeA_ScaleUsed := false
    modeA_EntryBar := na
    modeA_Entry := na
    modeA_SL := na
    modeA_TP1 := na
    modeA_TP2 := na
    modeA_TP3 := na
    modeA_BOS_Count := 0
    modeA_ScalePrice := na
    modeA_smcHL := na
    modeA_smcHH := na
    modeA_smcHH

// ===============================
// 🔁 RE-ENTRY RULE (SMC)
// ===============================
if structureDemand and not modeA_Valid and barstate.isconfirmed
    modeA_Locked := false
    modeA_Locked

// ===============================
// 🎯 ENTRY 1 — BOS / CHOCH DEMAND
// ===============================
if structureDemand and barstate.isconfirmed and not modeA_Valid and not modeA_Locked and not na(demandBottom) and not na(smcHL) and not na(smcHH)

    modeA_Valid := true
    modeA_Locked := true
    modeA_ScaleUsed := false

    modeA_EntryBar := bar_index
    modeA_Entry := math.round(close)

    // Gunakan 'demandBottom' (101) sebagai Anchor Fibonacci, bukan SL (100)
    modeA_smcHL := demandBottom
    modeA_smcHH := smcHH
    modeA_SL := math.round(demandBottom - 1) // SL tetap di bawah
    modeA_BOS_Count := 1
    modeA_BOS_Count

if modeA_Valid and not na(modeA_smcHL) and not na(modeA_smcHH)

    // --- KALKULASI FIBONACCI LOGARITMIK ---
    float logLow = math.log(modeA_smcHL)
    float logHigh = math.log(modeA_smcHH)
    float logDiff = logHigh - logLow

    float risk = math.abs(modeA_Entry - modeA_SL)

    if useRR_TP
        modeA_TP1 := math.round(modeA_Entry + risk * tp1_rr)
        modeA_TP2 := math.round(modeA_Entry + risk * tp2_rr)
        modeA_TP3 := math.round(modeA_Entry + risk * tp3_rr)
        modeA_TP3
    else // Menghitung TP dengan basis Logaritmik agar sesuai tampilan Chart
        modeA_TP1 := math.round(math.exp(logLow + tp1_mult * logDiff))
        modeA_TP2 := math.round(math.exp(logLow + tp2_mult * logDiff))
        modeA_TP3 := math.round(math.exp(logLow + tp3_mult * logDiff))
        modeA_TP3

    // ===============================
    // 🛡️ AUTO-KILLSWITCH: FILTER RISK TO REWARD
    // ===============================
    float riskDist = math.max(math.abs(modeA_Entry - modeA_SL), 0.0001)
    float current_tp2RR = math.abs(modeA_TP2 - modeA_Entry) / riskDist

    // Jika rasio TP2 berhimpitan atau kurang dari 0.60 RR, BATALKAN setup secara otomatis
    if current_tp2RR < 0.60
        modeA_Valid := false
        modeA_Locked := false
        modeA_EntryBar := na
        modeA_Entry := na
        modeA_SL := na
        modeA_TP1 := na
        modeA_TP2 := na
        modeA_TP3 := na
        modeA_smcHL := na
        modeA_smcHH := na
        modeA_smcHH


// ===============================
// 🎯 TP1 VISIBILITY (DESK MODE)
// ===============================
float tp1RR = math.abs(modeA_TP1 - modeA_Entry) / math.abs(modeA_Entry - modeA_SL)
bool tp1Visible = modeA_Valid and tp1RR >= 0.30

// ===============================
// ➕ BOS COUNT
// ===============================
if modeA_Valid and structureDemand and barstate.isconfirmed and bar_index > modeA_EntryBar
    modeA_BOS_Count := modeA_BOS_Count + 1
    modeA_BOS_Count

// ===============================
// 🟢 TP1 HIT VALIDATION
// ===============================
var bool tp1Hit = false
if not modeA_Valid
    tp1Hit := false
    tp1Hit
if modeA_Valid and not tp1Hit and high >= modeA_TP1
    tp1Hit := true
    tp1Hit

// ===============================
// 🟢 TP2 & TP3 HIT VALIDATION (HISTORICAL LOCK)
// ===============================
var bool tp2Hit = false
var bool tp3HitFlag = false

if not modeA_Valid
    tp2Hit := false
    tp3HitFlag := false
    tp3HitFlag

if modeA_Valid and not tp2Hit and high >= modeA_TP2
    tp2Hit := true
    tp2Hit

if modeA_Valid and not tp3HitFlag and high >= modeA_TP3
    tp3HitFlag := true
    tp3HitFlag

// ===============================
// 🔶 DANGER AREA (RISK ±25%) — VISUAL SELALU AKTIF
// ===============================
var box dangerBox = na
bool inDangerArea = false

if modeA_Valid
    float risk = math.abs(modeA_Entry - modeA_SL)
    float dr = risk * 0.25
    float top = modeA_Entry + dr
    float bot = modeA_Entry - dr

    inDangerArea := close <= top and close >= bot

    if na(dangerBox)
        dangerBox := box.new(modeA_EntryBar, top, bar_index + 3, bot, border_color = color.new(color.orange, 0), bgcolor = color.new(color.orange, 85))
        dangerBox
    else
        box.set_right(dangerBox, bar_index + 3)
        box.set_top(dangerBox, top)
        box.set_bottom(dangerBox, bot)

if not modeA_Valid and not na(dangerBox)
    box.delete(dangerBox)
    dangerBox := na
    dangerBox

// ===============================
// 🚨 SUPPLY NEAR EX
// ===============================
bool supplyNearEX = false
if modeA_Valid and array.size(bearBoxBottoms) > 0
    float risk = math.abs(modeA_Entry - modeA_SL)
    float dr = risk * 0.25
    for i = 0 to array.size(bearBoxBottoms) - 1 by 1
        if math.abs(array.get(bearBoxBottoms, i) - modeA_Entry) <= dr
            supplyNearEX := true
            supplyNearEX

// ===============================
// ➕ SCALE IN (MAX 1x) — OPSI A (AMAN)
// ===============================
bool canScaleIn = modeA_Valid and not modeA_ScaleUsed and not tp3Hit and structureDemand and barstate.isconfirmed and close > modeA_Entry and bar_index > modeA_EntryBar

if canScaleIn
    modeA_ScaleUsed := true
    modeA_ScalePrice := math.round(close)
    modeA_BOS_Count := modeA_BOS_Count + 1

    label.new(bar_index, high, 'AVG UP', style = label.style_label_down, color = color.new(color.blue, 70), textcolor = color.white, size = size.small)

// ===============================
// 🧠 CONFIDENCE ENGINE (DESK-GRADE 1–5 ★)
// ===============================
int confScore = modeA_BOS_Count
