var grpSwing = "Swing Detection"
pivotLen     = input.int(5,   "Pivot Length",       minval=2, maxval=50,   group=grpSwing, tooltip="The number of bars on each side of a high or low to confirm a pivot point.")
minStrength  = input.float(0.1, "Min ATR Strength", minval=0.0, maxval=5.0, step=0.05, group=grpSwing, tooltip="Minimum distance required between the pivot and its neighbors.")
maxAgeBars   = input.int(300, "Max Level Age (bars)", minval=20, maxval=2000, group=grpSwing, tooltip="Maximum bars a level exists before removal. Also used to calculate longevity scaling.")

var grpLevels = "Levels"
showSupport  = input.bool(true,  "Show Support Levels",    group=grpLevels)
showResist   = input.bool(true,  "Show Resistance Levels", group=grpLevels)
maxLevels    = input.int(10,      "Max Active Levels Each", minval=1, maxval=20, group=grpLevels)
mergeThresh  = input.float(0.5,  "Merge Threshold (ATR x)", minval=0.0, maxval=3.0, step=0.05, group=grpLevels)

showBreakLbl = input.bool(true,  "Show Break Labels",           group=grpBreak)
maxBroken    = input.int(7,      "Max Broken Levels Shown",    minval=0, maxval=20, group=grpBreak)

var grpVis = "Visuals"

srResColor   = input.color(color.new(#ff6f43, 0),  "Resistance Color",     group=grpVis)

lineWidth    = input.int(2, "Base Line Width",     minval=1, maxval=5,      group=grpVis, tooltip="Used when Longevity Scaling is OFF.")
useDynamicWidth = input.bool(false, "Enable Longevity Width", group=grpVis, tooltip="When enabled, older levels become thicker over time.")
maxLineWidth    = input.int(7, "Max Longevity Width", minval=1, maxval=10, group=grpVis, tooltip="The maximum thickness an old level can reach.")

type SRLevel
    float  price
    int    barStart
    int    levelType
    bool   active
    bool   broken
    int    breakBar
    bool   retested
    line   mainLine
    box    zoneBox
    label  breakLabel
    label  priceLabel

var array<SRLevel> activeLevels = array.new<SRLevel>()
var array<SRLevel> brokenLevels = array.new<SRLevel>()
var string         lastBreakDir = "—"
var int            lastBreakBar = na

atrVal  = ta.atr(14)
atrSafe = na(atrVal) or atrVal == 0 ? syminfo.mintick * 10 : atrVal

srPh = ta.pivothigh(high, pivotLen, pivotLen)
srPl = ta.pivotlow(low,   pivotLen, pivotLen)

f_isTooClose(_price, _type) =>
    bool tooClose = false
    int  sz = array.size(activeLevels)
    if sz > 0
        for i = 0 to sz - 1
            lvl = array.get(activeLevels, i)
            if lvl.levelType == _type and lvl.active
                if math.abs(lvl.price - _price) < mergeThresh * atrSafe
                    tooClose := true
                    break
    tooClose

f_addLevel(_price, _type, _barIdx) =>
    if not f_isTooClose(_price, _type)
        int sz  = array.size(activeLevels)
        int cnt = 0
        if sz > 0
            for i = 0 to sz - 1
                lvl = array.get(activeLevels, i)
                if lvl.levelType == _type and lvl.active
                    cnt += 1

        if cnt < maxLevels
            _baseColor = _type == 1 ? srResColor : srSupColor
            _zoneTop   = _price + zoneWidth * atrSafe * 0.5
            _zoneBot   = _price - zoneWidth * atrSafe * 0.5

            newLine = line.new(
                 x1     = _barIdx,
                 y1     = _price,
                 x2     = _barIdx + 1,
                 y2     = _price,
                 extend = extend.none,
                 color  = color.new(_baseColor, 0),
                 width  = lineWidth,
                 style  = line.style_solid
             )

            newBox = showSRZones ? box.new(
                 left         = _barIdx,
                 top          = _zoneTop,
                 right        = _barIdx + 1,
                 bottom       = _zoneBot,
                 border_color = color.new(_baseColor, 90),
                 bgcolor      = color.new(_baseColor, 88),
                 extend       = extend.none
             ) : na

            newPriceLbl = showPriceLbl ? label.new(
                 x         = bar_index + 20,
                 y         = _price,
                 text      = str.tostring(_price, format.mintick),
                 style     = label.style_label_center,
                 textcolor = _baseColor,
                 color     = color.new(chart.bg_color, 0),
                 size      = size.normal
             ) : na

            array.push(activeLevels, SRLevel.new(
                 price      = _price,
                 barStart   = _barIdx,
                 levelType  = _type,
                 active     = true,
                 broken     = false,
                 breakBar   = na,
                 retested   = false,
                 mainLine   = newLine,
                 zoneBox    = newBox,
                 breakLabel = na,
                 priceLabel = newPriceLbl
             ))

f_pivotStrong(_price, _type) =>
    bool strong = true
    if minStrength > 0
        if _type == 1
            nearHigh = math.max(high[pivotLen - 1], high[pivotLen + 1])
            if (_price - nearHigh) < minStrength * atrSafe
                strong := false
        else
            nearLow = math.min(low[pivotLen - 1], low[pivotLen + 1])
            if (nearLow - _price) < minStrength * atrSafe
                strong := false
    strong

int pruneSize = array.size(activeLevels)
if pruneSize > 0
    for i = pruneSize - 1 to 0
        if i < array.size(activeLevels)
            lvl = array.get(activeLevels, i)
            if lvl.active and (bar_index - lvl.barStart) > maxAgeBars
                line.delete(lvl.mainLine)
                if not na(lvl.zoneBox)
                    box.delete(lvl.zoneBox)
                if not na(lvl.priceLabel)
                    label.delete(lvl.priceLabel)
                array.remove(activeLevels, i)

int brokenPruneSize = array.size(brokenLevels)
if brokenPruneSize > 0
    for i = brokenPruneSize - 1 to 0
        if i < array.size(brokenLevels)
            lvl = array.get(brokenLevels, i)
            if (bar_index - lvl.breakBar) > maxAgeBars
                if not na(lvl.mainLine)
                    line.delete(lvl.mainLine)
                if not na(lvl.zoneBox)
                    box.delete(lvl.zoneBox)
                if not na(lvl.breakLabel)
                    label.delete(lvl.breakLabel)
                array.remove(brokenLevels, i)

if not na(srPh) and showResist
    if f_pivotStrong(srPh, 1)
        f_addLevel(srPh, 1, bar_index - pivotLen)

if not na(srPl) and showSupport
    if f_pivotStrong(srPl, -1)
        f_addLevel(srPl, -1, bar_index - pivotLen)

int activeSize = array.size(activeLevels)
if activeSize > 0
    for i = activeSize - 1 to 0
        if i >= array.size(activeLevels)
            continue
        lvl = array.get(activeLevels, i)
        if not lvl.active
            continue

        breakBuffer = breakSens * atrSafe

        if lvl.levelType == 1 and close > lvl.price + breakBuffer
            lvl.active   := false
            lvl.broken   := true
            lvl.breakBar := bar_index
            lastBreakDir := "Bullish"
            lastBreakBar := bar_index

            if not na(lvl.priceLabel)
                label.delete(lvl.priceLabel)
                lvl.priceLabel := na

            line.set_color(lvl.mainLine,  color.new(brokenColor, 0))
            line.set_style(lvl.mainLine,  line.style_dotted)
            line.set_width(lvl.mainLine,  1)
            line.set_extend(lvl.mainLine, extend.none)
            line.set_x2(lvl.mainLine,     bar_index)

            if not na(lvl.zoneBox)
                box.set_bgcolor(lvl.zoneBox,       color.new(brokenColor, 93))
                box.set_border_color(lvl.zoneBox,  color.new(brokenColor, 100))
                box.set_extend(lvl.zoneBox,        extend.none)
                box.set_right(lvl.zoneBox,         bar_index)
                mid = math.avg(lvl.zoneBox.get_top(), lvl.zoneBox.get_bottom())
                lvl.zoneBox.set_top(mid + breakBuffer)
                lvl.zoneBox.set_bottom(mid - breakBuffer)
                
            if showBreakLbl
                lvl.breakLabel := label.new(
                     x         = bar_index,
                     y         = lvl.price,
                     text      = "< Break",
                     style     = label.style_label_left,
                     textcolor = srSupColor,
                     color     = color.new(#22c55e, 100),
                     size      = size.normal
                )

            if showBroken
                array.push(brokenLevels, lvl)
                if array.size(brokenLevels) > maxBroken and array.size(brokenLevels) > 0
                    old = array.shift(brokenLevels)
                    if not na(old.mainLine)
                        line.delete(old.mainLine)
                    if not na(old.zoneBox)
                        box.delete(old.zoneBox)
                    if not na(old.breakLabel)
                        label.delete(old.breakLabel)
            else
                line.delete(lvl.mainLine)
                
                if not na(lvl.zoneBox)
                    box.delete(lvl.zoneBox)

            array.remove(activeLevels, i)

        else if lvl.levelType == -1 and close < lvl.price - breakBuffer
            lvl.active   := false
            lvl.broken   := true
            lvl.breakBar := bar_index
            lastBreakDir := "Bearish"
            lastBreakBar := bar_index

            if not na(lvl.priceLabel)
                label.delete(lvl.priceLabel)
                lvl.priceLabel := na

            line.set_color(lvl.mainLine,  color.new(brokenColor, 0))
            line.set_style(lvl.mainLine,  line.style_dotted)
            line.set_width(lvl.mainLine,  1)
            line.set_extend(lvl.mainLine, extend.none)
            line.set_x2(lvl.mainLine,     bar_index)

            if not na(lvl.zoneBox)
                box.set_bgcolor(lvl.zoneBox,       color.new(brokenColor, 93))
                box.set_border_color(lvl.zoneBox,  color.new(brokenColor, 100))
                box.set_extend(lvl.zoneBox,        extend.none)
                box.set_right(lvl.zoneBox,         bar_index)
                mid = math.avg(lvl.zoneBox.get_top(), lvl.zoneBox.get_bottom())
                lvl.zoneBox.set_top(mid + breakBuffer)
                lvl.zoneBox.set_bottom(mid - breakBuffer)
                
            if showBreakLbl
                lvl.breakLabel := label.new(
                     x         = bar_index,
                     y         = lvl.price,
                     text      = "< Break",
                     style     = label.style_label_left,
                     textcolor = srResColor,
                     color     = color.new(#ef4444, 100),
                     size      = size.normal
                )

            if showBroken
                array.push(brokenLevels, lvl)
                if array.size(brokenLevels) > maxBroken and array.size(brokenLevels) > 0
                    old = array.shift(brokenLevels)
                    if not na(old.mainLine)
                        line.delete(old.mainLine)
                    if not na(old.zoneBox)
                        box.delete(old.zoneBox)
                    if not na(old.breakLabel)
                        label.delete(old.breakLabel)
            else
                line.delete(lvl.mainLine)
                if not na(lvl.zoneBox)
                    box.delete(lvl.zoneBox)

            array.remove(activeLevels, i)

// }
// ＰＬＯＴ ――――――――――――――――――――――――――――――――――――――――――――――――――――――――{

int extSize = array.size(activeLevels)
if extSize > 0
    for i = 0 to extSize - 1
        lvl = array.get(activeLevels, i)
        if lvl.active
            if useDynamicWidth
                age = bar_index - lvl.barStart
                calcWidth = math.min(maxLineWidth, 1 + math.floor((age / maxAgeBars) * (maxLineWidth - 1)))

                line.set_width(lvl.mainLine, lineWidth)

            line.set_x2(lvl.mainLine, bar_index + 1)
            if not na(lvl.zoneBox)
                box.set_right(lvl.zoneBox, bar_index + 1)
            if showPriceLbl and not na(lvl.priceLabel)
                label.set_x(lvl.priceLabel, bar_index + 10)

// }

textSize = input.string("Medium", "Text Size", options=["Small", "Medium", "Large"])
labelSize = textSize == "Small" ? size.small : textSize == "Large" ? size.large : size.normal

labelOffset = input.int(20, "Label Right Offset", minval=0, maxval=100)

// ＴＲＡＣＫ ＣＵＲＲＥＮＴ ＰＥＲＩＯＤ ＬＥＶＥＬＳ ――――――――――――――――――――――{

var float currDayOpen = na
var float currDayHigh = na
var float currDayLow  = na

bool newDayDetected = dayofmonth != dayofmonth[1] or barstate.isfirst

if newDayDetected
    currDayOpen := open
    currDayHigh := high
    currDayLow  := low
else
    currDayHigh := math.max(nz(currDayHigh, high), high)
    currDayLow  := math.min(nz(currDayLow, low), low)

var float currWeekOpen = na
if weekofyear != weekofyear[1] or barstate.isfirst
    currWeekOpen := open

var float currMonthOpen = na
if month != month[1] or barstate.isfirst
    currMonthOpen := open

var float currYearOpen = na
if year != year[1] or barstate.isfirst
    currYearOpen := open

// }
// ＭＥＲＧＥ ＳＴＯＲＡＧＥ ――――――――――――――――――――――――――――――――――――――{

var float[] mergePrices = array.new_float()
var string[] mergeNames = array.new_string()
var color[] mergeColors = array.new_color()
var label[] allLabels = array.new<label>()

// DELETE OLD LABELS
if array.size(allLabels) > 0
    for i = 0 to array.size(allLabels) - 1
        label.delete(array.get(allLabels, i))
    array.clear(allLabels)

// CLEAR MERGE DATA
array.clear(mergePrices)
array.clear(mergeNames)
array.clear(mergeColors)

// }
// ＨＥＬＰＥＲ ＦＵＮＣＴＩＯＮ - ADD LEVEL TO MERGE ARRAYS ――――――――――{

addLevel(bool condition, float levelVal, string levelName, color txtColor) =>
    if condition and not na(levelVal)
        int idx = array.indexof(mergePrices, levelVal)
        
        if idx != -1
            string existingNames = array.get(mergeNames, idx)
            if not str.contains(existingNames, levelName)
                array.set(mergeNames, idx, existingNames + " / " + levelName)
        else
            array.push(mergePrices, levelVal)
            array.push(mergeNames, levelName)
            array.push(mergeColors, txtColor)

// }
// ＰＨＡＳＥ １: ＣＯＬＬＥＣＴ ＡＬＬ ＬＥＶＥＬＳ ＡＮＤ ＢＵＩＬＤ ＭＥＲＧＥ ＤＡＴＡ ――{

// Previous Day
addLevel(showPrevDayHigh, prevDayHigh, "Prev Day High", prevDayColor)
addLevel(showPrevDayLow, prevDayLow, "Prev Day Low", prevDayColor)
addLevel(showPrevDayClose, prevDayClose, "Prev Day Close", prevDayColor)
addLevel(showPrevDayMid, prevDayMid, "Prev Day Mid", prevDayColor)

// Previous Week
addLevel(showPrevWeekHigh, prevWeekHigh, "Prev Week High", prevWeekColor)
addLevel(showPrevWeekLow, prevWeekLow, "Prev Week Low", prevWeekColor)
addLevel(showPrevWeekClose, prevWeekClose, "Prev Week Close", prevWeekColor)
addLevel(showPrevWeekMid, prevWeekMid, "Prev Week Mid", prevWeekColor)

// Previous Month
addLevel(showPrevMonthHigh, prevMonthHigh, "Prev Month High", prevMonthColor)
addLevel(showPrevMonthLow, prevMonthLow, "Prev Month Low", prevMonthColor)
addLevel(showPrevMonthClose, prevMonthClose, "Prev Month Close", prevMonthColor)
addLevel(showPrevMonthMid, prevMonthMid, "Prev Month Mid", prevMonthColor)

// Previous Year
addLevel(showPrevYearHigh, prevYearHigh, "Prev Year High", prevYearColor)
addLevel(showPrevYearLow, prevYearLow, "Prev Year Low", prevYearColor)
addLevel(showPrevYearClose, prevYearClose, "Prev Year Close", prevYearColor)
addLevel(showPrevYearMid, prevYearMid, "Prev Year Mid", prevYearColor)
addLevel(showPrevYearOpen, prevYearOpen, "Prev Year Open", prevYearColor)

// Current Day
addLevel(showCurrDayOpen, currDayOpen, "Curr Day Open", currDayColor)
addLevel(showCurrDayHigh, currDayHigh, "Curr Day High", currDayColor)
addLevel(showCurrDayLow, currDayLow, "Curr Day Low", currDayColor)

// Current Week/Month/Year
addLevel(showCurrWeekOpen, currWeekOpen, "Curr Week Open", currWeekColor)
addLevel(showCurrMonthOpen, currMonthOpen, "Curr Month Open", currMonthColor)
addLevel(showCurrYearOpen, currYearOpen, "Curr Year Open", currYearColor)

// 4H
addLevel(show4hOpen, fourHOpen, "4 Hr Open", fourHColor)
addLevel(show4hPrevHigh, prev4hHigh, "Prev 4Hr High", fourHColor)
addLevel(show4hPrevLow, prev4hLow, "Prev 4Hr Low", fourHColor)

// }
// ＰＨＡＳＥ ２: ＣＲＥＡＴＥ ＬＡＢＥＬＳ ＦＲＯＭ ＭＥＲＧＥＤ ＤＡＴＡ ――{

int totalLevels = array.size(mergePrices)

for i = 0 to totalLevels - 1
    float price = array.get(mergePrices, i)
    string names = array.get(mergeNames, i)
    color txtColor = array.get(mergeColors, i)
    
    label newLabel = label.new(
        x = bar_index + labelOffset,
        y = price,
        text = names + "  " + str.tostring(price, format.mintick),
        xloc = xloc.bar_index,
        yloc = yloc.price,
        color = color.new(txtColor, 100),
        style = label.style_label_left,
        textcolor = txtColor,
        size = labelSize)
    
    array.push(allLabels, newLabel)
    
    if showLine
        line.new(
            bar_index,
            price,
            bar_index + 1,
            price,
            extend = extend.left,
            color = txtColor,
            style = lineStyle)
